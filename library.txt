DELIMITER $$
CREATE PROCEDURE Calculate_Fine(IN p_roll INT, IN p_bkn VARCHAR(50))
BEGIN
  DECLARE v_idate DATE;
  DECLARE v_days INT;
  DECLARE v_fineamt INT DEFAULT 0;


  SELECT Date_of_issue INTO v_idate 
  FROM A 
  WHERE Rollno = p_roll AND name_of_book = p_bkn;


  SET v_days = DATEDIFF(CURDATE(), v_idate);

  
  IF (v_days > 15) AND (v_days < 30) THEN
    SET v_fineamt = v_days * 5;
  ELSEIF (v_days >= 30) THEN
    SET v_fineamt = v_days * 10;
  END IF;


  INSERT INTO B VALUES (p_roll, CURDATE(), v_fineamt);


  UPDATE A SET Status='R' WHERE Rollno = p_roll AND name_of_book = p_bkn;


  SELECT 
    p_roll AS 'Roll_Number',
    p_bkn AS 'Book_Name',
    v_idate AS 'Issue_Date',
    v_days AS 'No_of_Days',
    v_fineamt AS 'FINE';

END$$
DELIMITER ;
